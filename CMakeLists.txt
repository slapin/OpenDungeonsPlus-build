cmake_minimum_required(VERSION 3.10)
project(ogre-base)
include(ExternalProject)
include(ProcessorCount)

set(OGRE_PATH "" CACHE PATH "Ogre path")
if (OGRE_PATH)
list(APPEND CMAKE_PREFIX_PATH OGRE_PATH)
endif()

set(CUSTOM_OPTS -DCMAKE_BUILD_TYPE=RelWithDebInfo -DCMAKE_PREFIX_PATH=${CMAKE_PREFIX_PATH} -DCMAKE_INSTALL_PREFIX=${CMAKE_CURRENT_SOURCE_DIR}/target -DBUILD_SHARED_LIBS=ON)
set(CUSTOM_OPTS_STATIC -DCMAKE_BUILD_TYPE=RelWithDebInfo -DCMAKE_PREFIX_PATH=${CMAKE_PREFIX_PATH} -DCMAKE_INSTALL_PREFIX=${CMAKE_CURRENT_SOURCE_DIR}/target-static -DBUILD_SHARED_LIBS=OFF)

set(OGRE_COMMON_OPTIONS
                -DOGRE_BUILD_RENDERSYSTEM_D3D9=FALSE
                -DOGRE_BUILD_RENDERSYSTEM_D3D11=FALSE
                -DOGRE_BUILD_RENDERSYSTEM_GL3PLUS=TRUE
                -DOGRE_BUILD_RENDERSYSTEM_GL=TRUE
                -DOGRE_BUILD_RENDERSYSTEM_GLES2=TRUE
                -DOGRE_BUILD_RENDERSYSTEM_METAL=FALSE
                -DOGRE_BUILD_RENDERSYSTEM_VULKAN=FALSE
                -DOGRE_BUILD_PLUGIN_GLSLANG=TRUE
                -DOGRE_BUILD_COMPONENT_OVERLAY=TRUE
                -DOGRE_BUILD_COMPONENT_BULLET=OFF
                -DOGRE_BUILD_PLUGIN_DOT_SCENE=TRUE
                -DOGRE_GLSUPPORT_USE_EGL:BOOL=FALSE
#               -DOGRE_INSTALL_DEPENDENCIES:BOOL=TRUE
#               -DASSIMP_BUILD_ASSIMP_TOOLS=OFF
#               -DOGRE_CONFIG_THREADS=3
#               -DOGRE_CONFIG_THREAD_PROVIDER="std"
)

ExternalProject_Add(pybind11
    GIT_REPOSITORY https://github.com/pybind/pybind11.git
    GIT_TAG        master
    SOURCE_DIR	   ${CMAKE_CURRENT_BINARY_DIR}/pybind
    CMAKE_ARGS     ${CUSTOM_OPTS} -DPYBIND11_TEST=OFF # Disable tests for faster build
)

ExternalProject_Add(ois
    GIT_REPOSITORY    https://github.com/wgois/OIS.git
    GIT_TAG           master
    SOURCE_DIR	      ${CMAKE_CURRENT_BINARY_DIR}/ois
    CMAKE_ARGS        ${CUSTOM_OPTS}
)

ExternalProject_Add(ogre
    GIT_REPOSITORY https://github.com/OGRECave/ogre.git
    GIT_TAG        master
    SOURCE_DIR	   ${CMAKE_CURRENT_BINARY_DIR}/ogre
    CMAKE_ARGS     ${CUSTOM_OPTS} ${OGRE_COMMON_OPTIONS}
                   -DOGRE_ASSERT_MODE=3
                   -DASSIMP_BUILD_ASSIMP_TOOLS=ON
                   -DOGRE_STATIC=FALSE
                   -DCMAKE_BUILD_TYPE=Debug

)

ExternalProject_Add(cegui
    GIT_REPOSITORY https://github.com/slapin/cegui.git
    GIT_TAG        v0-tomluchowski-14.x
    SOURCE_DIR	   ${CMAKE_CURRENT_BINARY_DIR}/cegui
    CMAKE_ARGS     ${CUSTOM_OPTS} -DCEGUI_BUILD_RENDERER_OGRE=ON
    		   -DCEGUI_BUILD_RENDERER_OPENGL=FALSE
		   -DCEGUI_BUILD_RENDERER_OPENGLES=FALSE 
		   -DCEGUI_BUILD_RENDERER_OPENGL3=FALSE
		   -DCEGUI_BUILD_RENDERER_OPENGLES=OFF -DCEGUI_BUILD_RENDERER_OPENGLES2=OFF
		   -DCEGUI_BUILD_RENDERER_OPENGLES3=OFF
		   -DCEGUI_BUILD_XMLPARSER_EXPAT=ON
		   -DCEGUI_BUILD_XMLPARSER_LIBXML2=OFF
		   -DCEGUI_SAMPLES_ENABLED=OFF
		   -DCEGUI_BUILD_IMAGECODEC_FREEIMAGE=FALSE
		   #-DCEGUI_OPTION_DEFAULT_IMAGECODEC=OgreRenderer-0.8
		   -DCEGUI_INSTALL_DEV_HEADERS=ON
    DEPENDS ogre
)

ExternalProject_Add(OpenDungeonsPlus
	SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/OpenDungeonsPlus
	CMAKE_ARGS
	${CUSTOM_OPTS} -DCMAKE_BUILD_TYPE=Debug -DOD_TREAT_WARNINGS_AS_ERRORS=OFF -DOD_DATA_PATH=./ -DOD_PLUGINS_CFG_PATH=./
	DEPENDS pybind11 ois ogre cegui
)
